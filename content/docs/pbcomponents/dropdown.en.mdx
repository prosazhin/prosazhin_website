## Usage

```tsx
import { UserCircleIcon } from '@heroicons/react/24/outline';
import { Badge, Dropdown } from '@prosazhin/pbcomponents';

<Dropdown>
  <Dropdown.Trigger>Dropdown Button</Dropdown.Trigger>
  <Dropdown.Content align='left'>
    <Dropdown.Item
      leftIcon={UserCircleIcon}
      badge={<Badge color='secondary'>2</Badge>}
    >
      My orders
    </Dropdown.Item>
    <Dropdown.Item>Profile setting</Dropdown.Item>
    <Dropdown.Item borderTop>Sign out</Dropdown.Item>
  </Dropdown.Content>
</Dropdown>;
```
