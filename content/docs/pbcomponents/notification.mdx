## Использование

```tsx
'use client';

import { Button, Notification } from '@prosazhin/pbcomponents';
import { useState } from 'react';

function Example() {
  const [open, setOpen] = useState(false);
  return (
    <>
      <Button onClick={() => setOpen(true)}>Show notification</Button>
      <Notification
        headline='Saved'
        open={open}
        onOpenChange={setOpen}
      >
        Changes saved.
      </Notification>
    </>
  );
}
```
