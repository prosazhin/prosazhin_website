## Usage

```tsx
import { Alert } from '@prosazhin/pbcomponents';

<Alert
  color='success'
  defaultOpen
>
  <Alert.Title>Saved</Alert.Title>
  <Alert.Description>Changes saved.</Alert.Description>
</Alert>;
```
