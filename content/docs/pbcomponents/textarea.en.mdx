## Usage

```tsx
'use client';

import { Textarea } from '@prosazhin/pbcomponents';
import { useState } from 'react';

function Example() {
  const [value, setValue] = useState('');
  return (
    <Textarea
      value={value}
      onChange={setValue}
      placeholder='Placeholder'
    />
  );
}
```
