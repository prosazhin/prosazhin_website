## Usage

```tsx
import { Container } from '@prosazhin/pbcomponents';

<Container size='full'>
  <Container.LeftAside>Left Aside</Container.LeftAside>
  <Container.Main>Container</Container.Main>
  <Container.RightAside>Right Aside</Container.RightAside>
</Container>;
```
