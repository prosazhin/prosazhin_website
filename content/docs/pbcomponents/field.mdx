Соберите `Field.Label`, `Field.Control`, `Field.Description` и `Field.Error`. `error` отмечает ошибочное состояние. Тексты `Field.Description` и `Field.Error` связываются с контролом через `aria-describedby`; собственный `error` контрола не затирается.

## Использование

```tsx
'use client';

import { Field, Input } from '@prosazhin/pbcomponents';
import { useState } from 'react';

function Example() {
  const [value, setValue] = useState('');
  return (
    <Field error={!value}>
      <Field.Label>Label</Field.Label>
      <Field.Control>
        <Input>
          <Input.Control
            value={value}
            onChange={setValue}
            placeholder='Placeholder'
          />
        </Input>
      </Field.Control>
      <Field.Description>Description</Field.Description>
      {!value && <Field.Error>Fill in the field</Field.Error>}
    </Field>
  );
}
```
