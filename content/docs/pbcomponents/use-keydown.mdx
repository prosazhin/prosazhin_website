Вызывает callback при нажатии указанных клавиш.

## Использование

```tsx
'use client';

import { useState } from 'react';
import { useKeydown } from '@prosazhin/pbcomponents';

function Example() {
  const [open, setOpen] = useState(true);
  useKeydown({ keys: ['Escape'], callback: () => setOpen(false) });
  return open ? <div>Press Escape</div> : null;
}
```
