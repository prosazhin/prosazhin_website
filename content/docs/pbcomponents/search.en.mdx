## Usage

```tsx
import { Search } from '@prosazhin/pbcomponents';

const options = ['One', 'Two', 'Three', 'Four', 'Five'].map((display, index) => ({
  display,
  value: String(index),
}));

<Search
  options={options}
  placeholder='Placeholder'
/>;
```
