`onChange` receives the next state, `value`, and the event. Use it for settings that take effect immediately. `description` adds text below the label.

## Usage

```tsx
'use client';

import { Toggle } from '@prosazhin/pbcomponents';
import { useState } from 'react';

function Example() {
  const [checked, setChecked] = useState(false);
  return (
    <Toggle
      value='one'
      checked={checked}
      onChange={setChecked}
    >
      Label
    </Toggle>
  );
}
```
