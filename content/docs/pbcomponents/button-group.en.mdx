## Usage

```tsx
import { Button, ButtonGroup, Dropdown } from '@prosazhin/pbcomponents';

<ButtonGroup size='m'>
  <Button theme='border'>One</Button>
  <Button theme='border'>Two</Button>
  <Dropdown>
    <Dropdown.Trigger theme='border'>Three</Dropdown.Trigger>
    <Dropdown.Content>
      <Dropdown.Item>One</Dropdown.Item>
      <Dropdown.Item>Two</Dropdown.Item>
      <Dropdown.Item>Three</Dropdown.Item>
    </Dropdown.Content>
  </Dropdown>
</ButtonGroup>;
```
