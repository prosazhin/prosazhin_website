## Usage

```tsx
'use client';

import { Checkbox } from '@prosazhin/pbcomponents';
import { useState } from 'react';

function Example() {
  const [checked, setChecked] = useState(false);
  return (
    <Checkbox
      value='one'
      checked={checked}
      onChange={setChecked}
    >
      Label
    </Checkbox>
  );
}
```
