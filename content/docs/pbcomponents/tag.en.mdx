## Usage

```tsx
'use client';

import { Tag } from '@prosazhin/pbcomponents';
import { useState } from 'react';

function Example() {
  const [checked, setChecked] = useState(false);
  return (
    <Tag
      checked={checked}
      onClick={() => setChecked(!checked)}
    >
      Tag
    </Tag>
  );
}
```
