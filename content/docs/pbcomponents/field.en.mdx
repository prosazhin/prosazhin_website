## Usage

```tsx
'use client';

import { Field, Input } from '@prosazhin/pbcomponents';
import { useState } from 'react';

function Example() {
  const [value, setValue] = useState('');
  return (
    <Field>
      <Field.Label>Label</Field.Label>
      <Field.Control>
        <Input>
          <Input.Control
            value={value}
            onChange={setValue}
            placeholder='Placeholder'
          />
        </Input>
      </Field.Control>
      <Field.Description>Description</Field.Description>
    </Field>
  );
}
```
