Calls a callback on a click or touch outside the provided elements.

## Usage

```tsx
'use client';

import { useRef, useState } from 'react';
import { useClickOutside } from '@prosazhin/pbcomponents';

function Example() {
  const [open, setOpen] = useState(true);
  const ref = useRef<HTMLDivElement>(null);
  useClickOutside({ refs: [ref], callback: () => setOpen(false) });
  return open ? <div ref={ref}>Menu</div> : null;
}
```
