Возвращает прогресс таймера от 0 до 100 с поддержкой паузы.

## Использование

```tsx
'use client';

import { useCountdown } from '@prosazhin/pbcomponents';

function Example() {
  const progress = useCountdown({ delay: 5000, paused: false });
  return (
    <progress
      value={progress}
      max={100}
    />
  );
}
```
